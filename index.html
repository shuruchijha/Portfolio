# ChemCalc — by Shuruchi
from chemcalc import FluidMechanics

# Reynolds number for pipe flow
pipe = FluidMechanics(
  fluid="crude_oil",
  density=850,  # kg/m³
  viscosity=0.004, # Pa·s
  velocity=2.5,   # m/s
  diameter=0.15   # m
)

print(f"Re = {pipe.reynolds():.0f}")
# Re = 79687 → Turbulent ✓

result = pipe.pressure_drop(L=100)
print(f"ΔP = {result:.2f} Pa")
